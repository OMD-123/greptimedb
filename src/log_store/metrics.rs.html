122    /// How long an object of the object store logstore takes to become durable
123    /// once its batch is sealed, sampled for every conditional create that
124    /// succeeds, an identical retry included. The time runs from the seal, so
125    /// it includes any wait for a create slot and, in the `enqueued` mode,
126    /// every repeated attempt and the delay before it.
127    ///
128    /// A create is a few milliseconds on a local object store and tens of
129    /// milliseconds on a cloud one, so the buckets start at 1ms and reach 8s.
130    pub static ref METRIC_OBJECT_STORE_WAL_SEAL_TO_DURABLE_SECONDS: Histogram = register_histogram!(
131        "greptime_logstore_object_store_wal_seal_to_durable_seconds",
132        "object store logstore seconds from sealing a batch to the success of its conditional create, slot waits and retries included",
133        exponential_buckets(0.001, 2.0, 14).unwrap(),
134    )
135    .unwrap();
136
137    /// How long an append of the object store logstore waits for its
138    /// acknowledgement in the `durable` mode, which is the write latency a
139    /// caller sees.
140    ///
141    /// The wait is one seal plus every earlier object plus one create, so with
142    /// the default 100ms flush interval it sits in the tens of milliseconds;
143    /// the buckets straddle that and leave room for a stalled uploader.
144    pub static ref METRIC_OBJECT_STORE_WAL_APPEND_ACK_SECONDS: Histogram = register_histogram!(
145        "greptime_logstore_object_store_wal_append_ack_seconds",
146        "object store logstore seconds an append waits for its acknowledgement in the durable mode",
147        exponential_buckets(0.002, 2.0, 14).unwrap(),
148    )
149    .unwrap();

151    /// How large the objects of the object store logstore are, which says
152    /// whether batches seal on size or on the flush interval.
153    ///
154    /// The buckets are powers of two from 1KiB, the size of a timer-sealed
155    /// batch of one small entry, to 16MiB, twice the 8MiB default batch size,
156    /// which is a threshold rather than an upper bound.
157    pub static ref METRIC_OBJECT_STORE_WAL_OBJECT_BYTES: Histogram = register_histogram!(
158        "greptime_logstore_object_store_wal_object_bytes",
159        "object store logstore bytes of each object sealed",
160        exponential_buckets(1024.0, 2.0, 15).unwrap(),
161    )
162    .unwrap();

164    /// How many entries the objects of the object store logstore hold, which
165    /// says how many writes one request amortises.

174    )

277    /// How long such an append stays held back, until it is admitted or until
278    /// it is refused, a stop, a terminal error or the teardown of the store

287        exponential_buckets(0.005, 3.0, 9).unwrap(),
288    )
289    .unwrap();
